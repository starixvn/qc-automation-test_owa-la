submit filters for sportsbook/number game
    [Arguments]    ${licensee}    ${currency}    ${basecurrency}    ${from_date}    ${to_date}    ${draw type}
    [Documentation]    This actions is used to submit filters for Sportsbook Win Loss report and Number Game Win Loss report.
    ...
    ...    ${draw type} = *'Normal'* or *'No Draw'* or *'No Draw for Turnover only'*
    ...
    ...    Example:
    ...
    ...    submit filters for sportsbook/number game | 12Bet | All | GBP | 2015-05-20 | 2015-05-25 | Normal
    # select licensee
    Select From List By Label    ${lstLicensee}    ${licensee}
    # select currency
    Select From List By Label    ${lstCurrency}    ${currency}
    # select Display total value in

    # select date range
    Run Keyword If    '${from_date}' != 'None' and '${to_date}'!= 'None'    select date range    ${from_date}    ${to_date}
    # select Type
    Run Keyword If    '${draw type}' != 'None'    select draw type    ${draw type}
    # click submit
    Click Element    ${btnSubmit}
